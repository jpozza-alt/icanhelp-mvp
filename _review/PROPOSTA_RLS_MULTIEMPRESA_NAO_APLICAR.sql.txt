-- Modelo de operacao transacional unica. Evita bypass direto e disputa entre
-- duas requisicoes simultaneas para o primeiro cadastro no mesmo tenant.
CREATE OR REPLACE FUNCTION public.icanhelp_nr1_create_company(
  p_tenant_id uuid,
  p_legal_name text,
  p_trade_name text,
  p_cnpj text,
  p_cnae_main text,
  p_company_size text,
  p_risk_grade text,
  p_employee_count integer,
  p_has_cipa boolean,
  p_has_sesmt boolean,
  p_has_public_service boolean,
  p_has_remote_work boolean,
  p_has_third_parties boolean,
  p_has_external_activities boolean,
  p_status text
)
RETURNS SETOF public.nr1_companies
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_actor uuid := auth.uid();
  v_company_exists boolean := false;
  v_multi_allowed boolean := false;
BEGIN
  IF v_actor IS NULL OR p_tenant_id IS NULL THEN
    RAISE EXCEPTION 'nr1_company_create_forbidden' USING ERRCODE = '42501';
  END IF;

  IF NOT EXISTS (
    SELECT 1 FROM public.tenant_memberships AS tm
    WHERE tm.tenant_id = p_tenant_id
      AND tm.user_id = v_actor
      AND tm.role IN ('owner', 'admin')
  ) THEN
    RAISE EXCEPTION 'nr1_company_create_forbidden' USING ERRCODE = '42501';
  END IF;

  -- Lock por tenant durante a transacao: duas criações concorrentes NAO podem
  -- passar ambas pelo teste 'nenhuma empresa existente'.
  PERFORM pg_catalog.pg_advisory_xact_lock(
    pg_catalog.hashtextextended(p_tenant_id::text, 141901)
  );

  -- Conta inclusive registros excluidos logicamente: nao pode apagar e criar
  -- novas empresas para contornar o limite contratual. Eventual reabertura
  -- deve ser fluxo explicito e auditavel, fora deste RPC.
  SELECT EXISTS (
    SELECT 1 FROM public.nr1_companies nc WHERE nc.tenant_id = p_tenant_id
  ) INTO v_company_exists;

  IF v_company_exists THEN
    SELECT EXISTS (
      SELECT 1
      FROM public.tenant_subscriptions ts
      JOIN public.subscription_plans sp
        ON sp.id = ts.subscription_plan_id
      WHERE ts.tenant_id = p_tenant_id
        AND ts.status IN ('active', 'trial', 'past_due')
        AND sp.is_active = TRUE
        AND sp.slug IN ('partner_sst', 'pasini')
        AND EXISTS (
          SELECT 1
          FROM public.plan_features pf
          JOIN public.features f ON f.id = pf.feature_id
          WHERE pf.subscription_plan_id = sp.id
            AND pf.is_enabled = TRUE
            AND f.is_active = TRUE
            AND f.feature_key = 'multi_company_portfolio'
        )
        AND EXISTS (
          SELECT 1
          FROM public.plan_features pf
          JOIN public.features f ON f.id = pf.feature_id
          WHERE pf.subscription_plan_id = sp.id
            AND pf.is_enabled = TRUE
            AND f.is_active = TRUE
            AND f.feature_key = 'partner_mode'
        )
    ) INTO v_multi_allowed;

    IF NOT v_multi_allowed THEN
      RAISE EXCEPTION 'nr1_company_create_forbidden' USING ERRCODE = '42501';
    END IF;
  END IF;

  IF p_legal_name IS NULL OR length(btrim(p_legal_name)) < 3 THEN
    RAISE EXCEPTION 'invalid_legal_name' USING ERRCODE = '22023';
  END IF;

  RETURN QUERY
    INSERT INTO public.nr1_companies (
      tenant_id, legal_name, trade_name, cnpj, cnae_main,
      company_size, risk_grade, employee_count,
      has_cipa, has_sesmt, has_public_service,
      has_remote_work, has_third_parties, has_external_activities,
      status, created_by
    ) VALUES (
      p_tenant_id, btrim(p_legal_name), p_trade_name, p_cnpj, p_cnae_main,
      p_company_size, p_risk_grade, p_employee_count,
      COALESCE(p_has_cipa, FALSE), COALESCE(p_has_sesmt, FALSE),
      COALESCE(p_has_public_service, FALSE), COALESCE(p_has_remote_work, FALSE),
      COALESCE(p_has_third_parties, FALSE), COALESCE(p_has_external_activities, FALSE),
      COALESCE(p_status, 'incomplete'), v_actor
    ) RETURNING *;
END;
$function$;

-- SECURITY DEFINER: conceder RPC apenas para sessoes autenticadas; o RPC
-- revalida usuario, tenant e direito dentro da transacao.
REVOKE ALL ON FUNCTION public.icanhelp_nr1_create_company(
  uuid,text,text,text,text,text,text,integer,boolean,boolean,boolean,
  boolean,boolean,boolean,text
) FROM PUBLIC;
GRANT EXECUTE ON FUNCTION public.icanhelp_nr1_create_company(
  uuid,text,text,text,text,text,text,integer,boolean,boolean,boolean,
  boolean,boolean,boolean,text
) TO authenticated;

-- Usuarios comuns nao podem inserir empresa diretamente via PostgREST.
REVOKE INSERT ON public.nr1_companies FROM PUBLIC, anon, authenticated;
DROP POLICY IF EXISTS nr1_companies_insert_member ON public.nr1_companies;

-- Impedir autoatribuicao de plano Partner SST via escrita direta.
-- COBRANCA: fluxos legitimamente internos devem escrever via service_role,
-- nunca confiando no papel owner/admin do proprio tenant.
REVOKE INSERT, UPDATE, DELETE ON public.tenant_subscriptions
  FROM PUBLIC, anon, authenticated;
DROP POLICY IF EXISTS tenant_subscriptions_insert_admin ON public.tenant_subscriptions;
DROP POLICY IF EXISTS tenant_subscriptions_update_admin ON public.tenant_subscriptions;
DROP POLICY IF EXISTS tenant_subscriptions_delete_admin ON public.tenant_subscriptions;

-- A validação dos GRANTs efetivos, policy restante, autenticação no RPC,
-- cenarios de concorrencia e as operacoes do sistema deve anteceder COMMIT.
-- O gatilho protetor no inicio faz esta proposta falhar intencionalmente.
COMMIT;
